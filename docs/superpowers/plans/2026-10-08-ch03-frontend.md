# EvalFlow Ch03 前端 Implementation Plan

> **For agentic workers:** 按 task 顺序逐个执行。步骤用 checkbox（`- [ ]`）追踪。每个 task 完成即验证通过；提交由用户指示（不自动 commit）。

**Goal:** 前端可视化评测平台:建评测集、发起 run、看三层指标 + 分桶聚合 + 逐条 trace,带路由/antd/测试/兜底/骨架屏。

**Architecture:** React + Vite + TS(独立 `frontend/`),react-router + antd + recharts;通过 CORS 调 FastAPI;后端补只读接口(列表带 dataset_name / 评测集详情)+ CORS + created_at/updated_at。

**Tech Stack:** React 18, Vite, TypeScript, react-router-dom, antd v5, recharts, axios, Vitest + Testing Library;后端复用 FastAPI + SQLite。

**Spec:** `docs/superpowers/specs/2026-10-08-ch03-frontend-design.md`

## Global Constraints

- 前端栈定死:React + Vite + TS + react-router-dom + antd + recharts,不加状态库(useState/useRequest 够)。
- 跨域用 FastAPI CORS 放行 `http://localhost:5173`(不用 Vite proxy)。
- 兜底四件套统一:骨架屏(Skeleton)/错误态(Result+重试)/空态(Empty)/渲染异常(ErrorBoundary),基座抽 `useRequest`。
- 后端补接口保持「只读、倒序、轻量」;评测集列表用后端分页(page/page_size + search),`item_count` 用 GROUP BY 聚合避免 N+1。不引入鉴权。
- 每个 task 完成即验证通过;是否提交由用户指示。

```python
from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173"],
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/runs")
def list_runs():
    s = SessionLocal()
    runs = s.query(Run).order_by(Run.id.desc()).all()
    # 一次 IN 查询回填评测集名称(问题4),避免逐条查的 N+1
    names = dict(s.query(Dataset.id, Dataset.name).filter(
        Dataset.id.in_({r.dataset_id for r in runs}))) if runs else {}
    s.close()
    return [{"id": r.id, "dataset_id": r.dataset_id, "dataset_name": names.get(r.dataset_id),
             "status": r.status, "created_at": r.created_at} for r in runs]

@app.get("/datasets/{dataset_id}")
def get_dataset(dataset_id: int):
    s = SessionLocal()
    d = s.get(Dataset, dataset_id)
    if d is None:
        s.close()
        raise HTTPException(status_code=404, detail="评测集不存在")
    count = s.query(DatasetItem).filter(DatasetItem.dataset_id == dataset_id).count()
    s.close()
    return {"id": d.id, "name": d.name, "description": d.description, "item_count": count}

@app.get("/datasets/{dataset_id}/items")
def list_items(dataset_id: int):
    s = SessionLocal()
    items = s.query(DatasetItem).filter(DatasetItem.dataset_id == dataset_id).all()
    s.close()
    return [{"id": it.id, "input": it.input, "expected_output": it.expected_output, "bucket": it.bucket} for it in items]

@app.post("/datasets/{dataset_id}/items")
def add_item(dataset_id: int, body: ItemIn):
    s = SessionLocal()
    it = DatasetItem(dataset_id=dataset_id, input=body.input,
                     expected_output=body.expected_output, bucket=body.bucket)
    s.add(it)
    ds = s.get(Dataset, dataset_id)
    if ds is not None:            # 加样本视为「修改」,刷新 updated_at(问题1)
        ds.updated_at = _now()
    s.commit()
    s.refresh(it)
    s.close()
    return {"id": it.id, "input": it.input, "bucket": it.bucket}

@app.get("/datasets")
def list_datasets(page: int = 1, page_size: int = 10, search: str = ""):
    """评测集列表(后端分页)。search 按名称模糊过滤,供「发起评测」下拉远程搜索。"""
    s = SessionLocal()
    q = s.query(Dataset)
    if search:
        q = q.filter(Dataset.name.contains(search))
    total = q.count()
    ds = q.order_by(Dataset.updated_at.desc()).offset((page - 1) * page_size).limit(page_size).all()  # 修改时间倒序(问题1)
    # item_count 用一次 GROUP BY 聚合,避免逐条 COUNT 的 N+1(列表 N 行就 N 次查询)
    counts = dict(
        s.query(DatasetItem.dataset_id, func.count(DatasetItem.id))
         .filter(DatasetItem.dataset_id.in_([d.id for d in ds]))
         .group_by(DatasetItem.dataset_id).all()
    ) if ds else {}
    items = [{"id": d.id, "name": d.name, "description": d.description,
              "item_count": counts.get(d.id, 0), "updated_at": d.updated_at} for d in ds]
    s.close()
    return {"items": items, "total": total}
```

```ts
// src/api/client.ts —— 统一请求封装(axios 实例 + 拦截器):请求前参数注入 / 请求后错误码兼容
import axios, { AxiosError, type AxiosRequestConfig } from "axios";

const BASE = import.meta.env.VITE_API_BASE ?? "http://localhost:8000";

// 统一错误类型:带状态码,上层可按 status 分支
export class ApiError extends Error {
  status: number;
  constructor(status: number, message: string) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

// 错误码 → 友好文案(后端是 FastAPI,错误体形如 {"detail": "..."})
const ERROR_MESSAGES: Record<number, string> = {
  400: "请求参数有误",
  401: "未登录或登录已过期",
  403: "没有权限",
  404: "资源不存在",
  422: "参数校验失败",
  500: "服务器开小差了",
};

// axios 实例:baseURL 统一,拦截器承载「请求前 / 请求后」两个钩子
const http = axios.create({ baseURL: BASE });

// 请求「前」钩子:统一参数注入(axios request 拦截器)
http.interceptors.request.use((config) => {
  config.headers.set("Content-Type", "application/json");
  // 以后要注入的公共参数都在这加,例如:
  // const token = getToken(); if (token) config.headers.set("Authorization", `Bearer ${token}`);
  // config.headers.set("X-Request-Id", crypto.randomUUID());
  return config;
});

// 请求「后」钩子:异常错误码兼容处理(axios response 拦截器,只处理 rejected)
http.interceptors.response.use(
  (resp) => resp,
  (error: AxiosError) => {
    // 主动取消(AbortController)原样抛出,交给 useRequest 判断;其余才是真错误
    if (error.code === "ERR_CANCELED") throw error;
    if (error.response) {
      const { status, data } = error.response;
      const detail = (data as { detail?: unknown } | null)?.detail;
      const msg = typeof detail === "string" ? detail : ERROR_MESSAGES[status] || `请求失败(${status})`;
      throw new ApiError(status, msg);
    }
    // 没有 response:网络层失败(断网/CORS/后端没起)
    throw new ApiError(0, "网络异常,请确认后端已启动");
  },
);

// 统一请求入口:所有接口都走这里(config 可带 signal 由 useRequest 传入;POST 用 data 传体,axios 自动 JSON.stringify)
export async function request<T>(path: string, config?: AxiosRequestConfig): Promise<T> {
  const resp = await http.request<T>({ url: path, ...config });
  if (resp.status === 204) return undefined as T;
  return resp.data;
}
```

**Step 3: useRequest.ts(统一 loading/error/retry 基座)**

```ts
import { useCallback, useEffect, useRef, useState } from "react";

// fn 接收 AbortSignal;deps 变化或组件卸载会自动打断上次调用(防 stale 覆盖)
export function useRequest<T>(fn: (signal: AbortSignal) => Promise<T>, deps: unknown[] = []) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const abortRef = useRef<AbortController | null>(null);

  const run = useCallback(() => {
    abortRef.current?.abort();               // 打断上一次未完成的调用
    const controller = new AbortController();
    abortRef.current = controller;

    setLoading(true);
    setError(null);
    fn(controller.signal)
      .then(setData)
      .catch((e) => {
        if (controller.signal.aborted) return; // 主动取消不算错误
        setError(e instanceof Error ? e.message : String(e));
      })
      .finally(() => {
        if (!controller.signal.aborted) setLoading(false);
      });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);

  useEffect(() => {
    run();
    return () => abortRef.current?.abort();  // 组件卸载时打断
  }, [run]);

  return { data, loading, error, reload: run };
}
```

**Step 4: ErrorBoundary.tsx(渲染异常兜底)**

```tsx
import { Component, type ReactNode } from "react";
import { Button, Result } from "antd";

export default class ErrorBoundary extends Component<{ children: ReactNode }, { error: Error | null }> {
  state = { error: null as Error | null };
  static getDerivedStateFromError(error: Error) { return { error }; }
  render() {
    if (this.state.error) {
      return <Result status="error" title="页面出错了" subTitle={String(this.state.error)}
        extra={<Button type="primary" onClick={() => this.setState({ error: null })}>重试</Button>} />;
    }
    return this.props.children;
  }
}
```

### Task 5: run 发起 + 列表(RunsView,antd + 轮询)

**Files:**
- Create: `src/views/RunsView.tsx`

**功能:** 选评测集发起 run(POST /runs),列表展示所有 run(状态 Tag + 评测集名称),有 running 就每 1.5s 轮询,「操作」列点「查看」跳详情。发起评测下拉走后端远程搜索(showSearch + search 参数,数据量大时不一次性下拉全量)。样本数为 0 的评测集在下拉里置灰并标注 `(空)`(问题5:空集禁止发起评测)。

```tsx
import { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";
import { Button, Result, Select, Skeleton, Space, Table, Tag } from "antd";
import { api } from "../api";
import { useRequest } from "../hooks/useRequest";
import type { Run } from "../types";

const STATUS: Record<string, { color: string; text: string }> = {
  running: { color: "gold", text: "运行中" },
  done: { color: "green", text: "完成" },
  failed: { color: "red", text: "失败" },
};

export default function RunsView() {
  const nav = useNavigate();
  const [kw, setKw] = useState("");
  // 发起评测下拉走后端远程搜索:输入即带 search 请求,不一次性下拉全量(数据量大时)
  const { data: datasets } = useRequest((s) => api.listDatasets({ page: 1, page_size: 20, search: kw }, s), [kw]);
  const { data: runs, loading, error, reload } = useRequest((s) => api.listRuns(s));
  const [sel, setSel] = useState<number | null>(null);

  useEffect(() => {
    const hasRunning = runs?.some((r) => r.status === "running");
    if (!hasRunning) return;
    const t = setInterval(reload, 1500);
    return () => clearInterval(t);
  }, [runs, reload]);

  if (loading) return <Skeleton active paragraph={{ rows: 6 }} />;
  if (error) return <Result status="error" title="加载失败" subTitle={error} extra={<Button onClick={reload}>重试</Button>} />;

  const start = async () => { if (sel === null) return; await api.createRun(sel); reload(); };

  return (
    <Space direction="vertical" style={{ width: "100%" }} size="large">
      <Space>
        <Select
          showSearch
          onSearch={setKw}
          filterOption={false}  // 交给后端按名称模糊过滤,不用前端 filterOption
          placeholder="选评测集" style={{ width: 240 }} value={sel ?? undefined}
          options={(datasets?.items ?? []).map((d) => ({
            value: d.id,
            label: (d.item_count ?? 0) === 0 ? `${d.name}(空)` : d.name,
            disabled: (d.item_count ?? 0) === 0,  // 空评测集禁止发起评测(问题5)
          }))} onChange={setSel} />
        <Button type="primary" disabled={sel === null} onClick={start}>发起评测</Button>
      </Space>
      <Table rowKey="id" dataSource={runs ?? []}
        columns={[
          { title: "ID", dataIndex: "id" },
          { title: "评测集", dataIndex: "dataset_name" },  // 问题4:显示评测集名称
          { title: "状态", dataIndex: "status", render: (s: string) => <Tag color={STATUS[s]?.color}>{STATUS[s]?.text ?? s}</Tag> },
          { title: "时间", dataIndex: "created_at" },
          { title: "操作", key: "action", render: (_: unknown, r: Run) => (
            <Button type="link" size="small" onClick={() => nav(`/runs/${r.id}`)}>查看</Button>
          )},
        ]} />
    </Space>
  );
}
```
